local character = player.Character or player.CharacterAdded:Wait()
local flying = false
local speed = 80

ToggleButton.MouseButton1Click:Connect(function()
    MainFrame.Visible = not MainFrame.Visible

local function getTool()
    if not character then return nil end
    return character:FindFirstChildOfClass("Tool")
end

-- الخيار الأول
TargetSlapButton.MouseButton1Click:Connect(function()
    allActive = not allActive
    if allActive then
        TargetSlapButton.Text = "جاري قذف الجميع..."
        TargetSlapButton.BackgroundColor3 = Color3.fromRGB(0, 100, 0)
        task.spawn(function()
            while allActive do
                task.wait(0.02)
                local tool = getTool()
                if tool then
                    local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildOfClass("Part")
                    if handle then
                        for _, p in pairs(game.Players:GetPlayers()) do
                            if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                                local targetHrp = p.Character.HumanoidRootPart
                                pcall(function()
                                    firetouchinterest(targetHrp, handle, 0)
                                    firetouchinterest(targetHrp, handle, 1)
                                    if not targetHrp:FindFirstChild("VoidForce") then
                                        local bv = Instance.new("BodyVelocity")
                                        bv.Name = "VoidForce"
                                        bv.MaxForce = Vector3.new(1, 1, 1) * 9e8
                                        bv.Velocity = Vector3.new(0, 3000, 0)
                                        bv.Parent = targetHrp
                                        game:GetService("Debris"):AddItem(bv, 0.05)
                                    end
                                end)
                            end
                        end
                    end
                end
            end
        end)
    else
        allActive = false
        TargetSlapButton.Text = "تطير الجميع خارج الماب"
        TargetSlapButton.BackgroundColor3 = Color3.fromRGB(40, 0, 0)
    end
end)

FlyButton.MouseButton1Click:Connect(function()
    flying = not flying
    if flying then
        FlyButton.Text = "طيران خارق + كسر الحواجز: مفعل ✈️"
        FlyButton.BackgroundColor3 = Color3.fromRGB(100, 0, 0)
        local torso = character:FindFirstChild("HumanoidRootPart")
        if not torso then return end
        
        local bg = Instance.new("BodyGyro", torso)
        bg.P = 9e4; bg.maxTorque = Vector3.new(9e5, 9e5, 9e5); bg.cframe = torso.CFrame
        local bv = Instance.new("BodyVelocity", torso)
        bv.velocity = Vector3.new(0, 0.1, 0); bv.maxForce = Vector3.new(9e5, 9e5, 9e5)
        
        task.spawn(function()
            local mouse = player:GetMouse()
            while flying and torso and character:FindFirstChild("Humanoid") do
                game:GetService("RunService").RenderStepped:Wait()
                bv.velocity = mouse.Hit.lookVector * speed
                bg.cframe = CFrame.new(torso.Position, mouse.Hit.p)
                
                for _, part in pairs(character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
            bv:Destroy(); bg:Destroy()
        end)
    else
        flying = false
        FlyButton.Text = "طيران خارق + كسر الحواجز: إيقاف"
        FlyButton.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
        
        for _, part in pairs(character:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.CanClone = true
                part.CanCollide = true
            end
        end
    end
end)

-- الخيار الخامس المُصحح: مضاد الصفع (Anti-Slap) الحقيقي لمنع الطرد أو التأثر بالصفعات
AntiSlapButton.MouseButton1Click:Connect(function()
    antiSlapActive = not antiSlapActive
    if antiSlapActive then
        AntiSlapButton.Text = "مضاد الصفع (Anti-Slap): مفعل 🛡️"
        AntiSlapButton.BackgroundColor3 = Color3.fromRGB(0, 100, 150)
        task.spawn(function()
            while antiSlapActive do
                task.wait(0.01)
                pcall(function()
                    local hrp = character and character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        -- منع تفعيل أو تثبيت السجن/التجميد وتفريغ أي قوة دفع مفاجئة تتسبب في طردك
                        if hrp.Anchored then
                            hrp.Anchored = false
                        end
                        -- مراقبة السرعة المتجهة وإلغاء أي طرد مفاجئ للخارج
                        if hrp.AssemblyLinearVelocity.Magnitude > 200 and not flying then
                            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                        end
                        -- الغاء تفعيل ملامسة أدوات الآخرين لشخصيتك عبر تعطيل الـ Touches الخارجية المضرّة إن وجدت
                        for _, p in pairs(game.Players:GetPlayers()) do
                            if p ~= player and p.Character then
                                for _, part in pairs(p.Character:GetDescendants()) do
                                    if part:IsA("BasePart") then
                                        part.CanTouch = false
                                    end
                                end
                            end
                        end
                    end
                end)
            end
        end)
    else
        antiSlapActive = false
        AntiSlapButton.Text = "مضاد الصفع (Anti-Slap): إيقاف"
        AntiSlapButton.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    end
end)
